    45→function scrollToAnchor(e: React.MouseEvent, href: string) {
    46→  e.preventDefault();
    47→  if (typeof document === "undefined") return;
    48→  // Guard against invalid selectors: "#", empty strings, or malformed hrefs.
    49→  // querySelector throws a SyntaxError for "#" alone.
    50→  if (!href || href === "#" || href.length < 2) return;
    51→  try {
    52→    const el = document.querySelector(href);
    53→    if (el) el.scrollIntoView({ behavior: "smooth", block: "start" });
    54→  } catch {
    55→    // Invalid selector — silently ignore rather than crash.
    56→  }
    57→}

   291→function Nav({ go }: { go: GoFn }) {
   292→  const [scrolled, setScrolled] = React.useState(false);
   293→
   294→  React.useEffect(() => {
   295→    if (typeof window === "undefined") return;
   296→    const onScroll = () => setScrolled(window.scrollY > 12);
   297→    onScroll();
   298→    window.addEventListener("scroll", onScroll, { passive: true });
   299→    return () => window.removeEventListener("scroll", onScroll);
   300→  }, []);
   301→
   302→  return (
   303→    <header
   304→      className={cn(
   305→        "fixed inset-x-0 top-0 z-50 transition-all duration-300",
   306→        scrolled
   307→          ? "border-b backdrop-blur-xl"
   308→          : "border-b border-transparent",
   309→      )}
   310→      style={
   311→        scrolled
   312→          ? {
   313→              background: "rgba(13, 14, 18, 0.72)",
   314→              borderColor: "var(--ld-border)",
   315→              backdropFilter: "blur(20px)",
   316→              WebkitBackdropFilter: "blur(20px)",
   317→            }
   318→          : undefined
   319→      }
   320→    >
   321→      <div className="mx-auto flex h-16 max-w-6xl items-center justify-between px-6">

   369→  const heroRef = React.useRef<HTMLElement | null>(null);
   370→
   371→  const onMouseMove = (e: React.MouseEvent) => {
   372→    const el = heroRef.current;
   373→    if (!el) return;
   374→    const rect = el.getBoundingClientRect();
   375→    const x = ((e.clientX - rect.left) / rect.width) * 100;
   376→    const y = ((e.clientY - rect.top) / rect.height) * 100;
   377→    el.style.setProperty("--mx", `${x}%`);
   378→    el.style.setProperty("--my", `${y}%`);
   379→  };
   380→
   381→  return (

   383→      ref={heroRef}
