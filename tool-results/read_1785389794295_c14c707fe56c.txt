    45→function scrollToAnchor(e: React.MouseEvent, href: string) {
    46→  e.preventDefault();
    47→  if (typeof document === "undefined") return;
    48→  // Guard against invalid selectors: "#", empty strings, or malformed hrefs.
    49→  // querySelector throws a SyntaxError for "#" alone.
    50→  if (!href || href === "#" || href.length < 2) return;
    51→  try {
    52→    const el = document.querySelector(href);
    53→    if (el) el.scrollIntoView({ behavior: "smooth", block: "start" });
    54→  } catch {
    55→    // Invalid selector — silently ignore rather than crash.
    56→  }
    57→}

   296→function Nav({ go }: { go: GoFn }) {
   297→  const [scrolled, setScrolled] = React.useState(false);
   298→
   299→  React.useEffect(() => {
   300→    if (typeof window === "undefined") return;
   301→    const onScroll = () => setScrolled(window.scrollY > 12);
   302→    onScroll();
   303→    window.addEventListener("scroll", onScroll, { passive: true });
   304→    return () => window.removeEventListener("scroll", onScroll);
   305→  }, []);
   306→
   307→  return (
   308→    <header
   309→      className={cn(
   310→        "fixed inset-x-0 top-0 z-50 transition-all duration-300",
   311→        scrolled
   312→          ? "border-b backdrop-blur-xl"
   313→          : "border-b border-transparent",
   314→      )}
   315→      style={
   316→        scrolled
   317→          ? {
   318→              background: "rgba(13, 14, 18, 0.72)",
   319→              borderColor: "var(--ld-border)",
   320→              backdropFilter: "blur(20px)",
   321→              WebkitBackdropFilter: "blur(20px)",
   322→            }
   323→          : undefined
   324→      }
   325→    >
   326→      <div className="mx-auto flex h-16 max-w-6xl items-center justify-between px-6">
   327→        <button
   328→          onClick={() => go("dashboard")}
   329→          className="flex items-center gap-2 rounded-md focus-ring"

   373→function Hero({ go }: { go: GoFn }) {
   374→  const heroRef = React.useRef<HTMLElement | null>(null);
   375→
   376→  const onMouseMove = (e: React.MouseEvent) => {
   377→    const el = heroRef.current;
   378→    if (!el) return;
   379→    const rect = el.getBoundingClientRect();
   380→    const x = ((e.clientX - rect.left) / rect.width) * 100;
   381→    const y = ((e.clientY - rect.top) / rect.height) * 100;
   382→    el.style.setProperty("--mx", `${x}%`);
   383→    el.style.setProperty("--my", `${y}%`);
   384→  };
   385→
   386→  return (
   387→    <section
   388→      ref={heroRef}
