    38→function scrollToAnchor(e: React.MouseEvent, href: string) {
    39→  e.preventDefault();
    40→  if (typeof document === "undefined") return;
    41→  // Guard against invalid selectors: "#", empty strings, or malformed hrefs.
    42→  // querySelector throws a SyntaxError for "#" alone.
    43→  if (!href || href === "#" || href.length < 2) return;
    44→  try {
    45→    const el = document.querySelector(href);
    46→    if (el) el.scrollIntoView({ behavior: "smooth", block: "start" });
    47→  } catch {
    48→    // Invalid selector — silently ignore rather than crash.
    49→  }
    50→}

   224→function Nav({ go }: { go: GoFn }) {
   225→  const [scrolled, setScrolled] = React.useState(false);
   226→
   227→  React.useEffect(() => {
   228→    if (typeof window === "undefined") return;
   229→    const onScroll = () => setScrolled(window.scrollY > 12);
   230→    onScroll();
   231→    window.addEventListener("scroll", onScroll, { passive: true });
   232→    return () => window.removeEventListener("scroll", onScroll);
   233→  }, []);
   234→
   235→  return (
   236→    <header
   237→      className={cn(
   238→        "fixed inset-x-0 top-0 z-50 transition-all duration-300",
   239→        scrolled
   240→          ? "border-b backdrop-blur-xl"
   241→          : "border-b border-transparent",
   242→      )}
   243→      style={
   244→        scrolled
   245→          ? {
   246→              background: "rgba(13, 14, 18, 0.72)",
   247→              borderColor: "var(--ld-border)",
   248→              backdropFilter: "blur(20px)",
   249→              WebkitBackdropFilter: "blur(20px)",
   250→            }
   251→          : undefined
   252→      }
   253→    >
   254→      <div className="mx-auto flex h-16 max-w-6xl items-center justify-between px-6">
   255→        <button
   256→          onClick={() => go("dashboard")}
   257→          className="flex items-center gap-2 rounded-md focus-ring"
   258→          aria-label="Lemniscate — open the reading room"
   259→        >
   260→          <FanLemniscateMark size={36} />
   261→          <span className="font-display text-xl tracking-tight">Lemniscate</span>
   262→        </button>
   263→
   264→        <nav className="hidden items-center gap-8 md:flex" aria-label="Primary">
