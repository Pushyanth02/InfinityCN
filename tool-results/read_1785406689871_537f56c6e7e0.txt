   425→   425→/**
   426→   426→ * Effective chunk count for a chapter. If the chapter has been OCR-refined,
   427→   427→ * we split the refined text into ~1200-char chunks at paragraph boundaries.
   428→   428→ * Otherwise we use the parser-produced chunks.
   429→   429→ */
   430→   430→function chapterChunkCount(ch: Chapter | undefined): number {
   431→   431→  if (!ch) return 0;
   432→   432→  if (ch.refinedText) {
   433→   433→    // Split refined text into chunks for focus-mode navigation
   434→   434→    const paras = ch.refinedText.split(/\n{2,}/).filter(Boolean);
   435→   435→    return Math.max(1, Math.ceil(paras.length / 3));
   436→   436→  }
   437→   437→  return ch.chunks.length;
   438→   438→}
   439→   439→
   440→   440→/** Split refined text into paragraph groups for rendering. */
   441→   441→function refinedTextGroups(text: string): string[] {
   442→   442→  const paras = text.split(/\n{2,}/).map((p) => p.trim()).filter(Boolean);
   443→   443→  const groups: string[] = [];
   444→   444→  for (let i = 0; i < paras.length; i += 3) {
   445→   445→    groups.push(paras.slice(i, i + 3).join("\n\n"));
   446→   446→  }
   447→   447→  return groups;
   448→   448→}

   512→   512→    if (!doc || chapters.length === 0) return;
   513→   513→    const lastGlobal = Math.max(
   514→   514→      0,
   515→   515→      Math.min(doc.lastChunkIndex ?? 0, totalChunks - 1),
   516→   516→    );
   517→   517→    let chapterIdx = 0;
   518→   518→    let chunkIdx = 0;
   519→   519→    let acc = 0;
   520→   520→    let placed = false;
   521→   521→    for (let i = 0; i < chapters.length; i++) {
   522→   522→      const len = chapterChunkCount(chapters[i]);
   523→   523→      if (acc + len > lastGlobal) {
   524→   524→        chapterIdx = i;
   525→   525→        chunkIdx = lastGlobal - acc;
   526→   526→        placed = true;
   527→   527→        break;
   528→   528→      }
   529→   529→      acc += len;
   530→   530→    }
   531→   531→    if (!placed) {
   532→   532→      chapterIdx = chapters.length - 1;
   533→   533→      chunkIdx = chapterChunkCount(chapters[chapters.length - 1]) - 1;
   534→   534→    }
   535→   535→    setCurrentChapterIndex(chapterIdx);
   536→   536→    setActiveChunkIndex(Math.max(0, chunkIdx));
   537→   537→    // Only on doc open — subsequent changes come from navigation.
   538→   538→  }, [doc?.id]);
   539→   539→
   540→   540→  // Clear refining state when chapter changes (the previous chapter's
   541→   541→  // refinement job is abandoned — its result will be ignored via the abort ref)
   542→   542→  useEffect(() => {
   543→   543→    setRefining(false);
   544→   544→    refineAbortRef.current = null;
   545→   545→  }, [currentChapterIndex]);
   546→   546→
   547→   547→  /* ---------- per-chapter OCR refinement (background, non-blocking) ----------
   548→   548→   * When a chapter is opened, it renders instantly with raw text. Then this
   549→   549→   * effect triggers OCR refinement in the background. When refinement
   550→   550→   * completes, the chapter's `refinedText` is updated and the reader
   551→   551→   * re-renders with the cleaned text. The refinement is persisted server-side
   552→   552→   * so the chapter is never reprocessed. */
   553→   553→  const currentChapterRefined = currentChapter?.refinedText;
   554→   554→  useEffect(() => {
   555→   555→    if (!documentId || !content || !currentChapter) return;
   556→   556→    const chIdx = currentChapterIndex;
   557→   557→
   558→   558→    // Already refined? Skip.
   559→   559→    if (currentChapterRefined) return;
   560→   560→    // Already triggered for this chapter? Skip.
   561→   561→    if (refineTriggered.has(chIdx)) return;
   562→   562→
   563→   563→    // Mark as triggered
   564→   564→    setRefineTriggered((prev) => new Set(prev).add(chIdx));
   565→   565→    setRefining(true);
   566→   566→    refineAbortRef.current = chIdx;
   567→   567→
   568→   568→    let cancelled = false;
   569→   569→    refineChapter(documentId, chIdx, false)
   570→   570→      .then((result) => {
   571→   571→        if (cancelled || refineAbortRef.current !== chIdx) return;
   572→   572→        if (result.refined && result.refinedText) {
   573→   573→          // Update the chapter's refinedText in local state
   574→   574→          setContent((prev) => {
   575→   575→            if (!prev) return prev;
   576→   576→            const newChapters = prev.chapters.map((ch, i) =>
   577→   577→              i === chIdx ? { ...ch, refinedText: result.refinedText } : ch,
   578→   578→            );
   579→   579→            return { ...prev, chapters: newChapters };
   580→   580→          });
   581→   581→        }
   582→   582→      })
   583→   583→      .catch(() => {
   584→   584→        // Best-effort — silently ignore (rate limits, network errors, etc.)
   585→   585→      })
   586→   586→      .finally(() => {
   587→   587→        if (cancelled || refineAbortRef.current !== chIdx) return;
   588→   588→        setRefining(false);
   589→   589→      });
   590→   590→
   591→   591→    // Safety timeout — don't show the indicator for more than 60 seconds
   592→   592→    const safetyTimer = setTimeout(() => {
   593→   593→      if (!cancelled && refineAbortRef.current === chIdx) {
   594→   594→        setRefining(false);
   595→   595→      }
   596→   596→    }, 60_000);
   597→   597→
   598→   598→    return () => {
   599→   599→      cancelled = true;
   600→   600→      clearTimeout(safetyTimer);
   601→   601→    };
   602→   602→  }, [documentId, currentChapterIndex, currentChapterRefined, content, refineTriggered]);
   603→   603→
   604→   604→  /* ---------- scroll active chunk into view ---------- */
   605→   605→  const scrollRafRef = useRef<number | null>(null);
   606→   606→  useEffect(() => {
   607→   607→    if (scrollRafRef.current) cancelAnimationFrame(scrollRafRef.current);
   608→   608→    scrollRafRef.current = requestAnimationFrame(() => {
   609→   609→      if (!currentChapter) return;
   610→   610→      const el = document.querySelector(
   611→   611→        `.reader-article [data-ordinal="${activeChunkIndex}"]`,
   612→   612→      );
   613→   613→      if (el) {
   614→   614→        el.scrollIntoView({ behavior: "smooth", block: "center" });
   615→   615→      }
   616→   616→    });
   617→   617→    return () => {
   618→   618→      if (scrollRafRef.current) cancelAnimationFrame(scrollRafRef.current);
   619→   619→    };
   620→   620→  }, [currentChapterIndex, activeChunkIndex, currentChapter]);
   621→   621→
   622→   622→  /* ---------- debounced progress patch ---------- */
   623→   623→  // Use refs to avoid a feedback loop: patching updates `doc`, which would
   624→   624→  // otherwise re-fire this effect and patch again.
   625→   625→  const patchTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   626→   626→  const sessionMarkedReadRef = useRef(false);
   627→   627→  useEffect(() => {
   628→   628→    sessionMarkedReadRef.current = sessionMarkedRead;
   629→   629→  }, [sessionMarkedRead]);
   630→   630→  useEffect(() => {
   631→   631→    if (!documentId || totalChunks === 0) return;
   632→   632→    if (patchTimer.current) clearTimeout(patchTimer.current);
   633→   633→    patchTimer.current = setTimeout(async () => {
   634→   634→      const patch: Partial<{
   635→   635→        readingProgress: number;
   636→   636→        lastChunkIndex: number;
   637→   637→        lastReadAt: string;
   638→   638→      }> = {
   639→   639→        readingProgress,
   640→   640→        lastChunkIndex: globalChunkIndex,
   641→   641→      };
   642→   642→      if (!sessionMarkedReadRef.current) {
   643→   643→        patch.lastReadAt = new Date().toISOString();
   644→   644→        setSessionMarkedRead(true);
   645→   645→      }
   646→   646→      try {
   647→   647→        const updated = await patchDocument(documentId, patch);
   648→   648→        setDoc((prev) => (prev ? { ...prev, ...updated } : prev));
   649→   649→      } catch {
   650→   650→        // ignore network hiccups
   651→   651→      }
   652→   652→    }, 600);
   653→   653→    return () => {
   654→   654→      if (patchTimer.current) clearTimeout(patchTimer.current);
   655→   655→    };
   656→   656→  }, [
   657→   657→    globalChunkIndex,
   658→   658→    readingProgress,
   659→   659→    documentId,
   660→   660→    totalChunks,
   661→   661→    setDoc,
   662→   662→  ]);
   663→   663→
   664→   664→  /* ---------- navigation ---------- */
   665→   665→  const nextChunk = useCallback(() => {
   666→   666→    if (!currentChapter || chapters.length === 0) return;
   667→   667→    if (activeChunkIndex < chapterChunkCount(currentChapter) - 1) {
   668→   668→      setActiveChunkIndex((i) => i + 1);
   669→   669→    } else if (currentChapterIndex < chapters.length - 1) {
   670→   670→      setCurrentChapterIndex((i) => i + 1);
   671→   671→      setActiveChunkIndex(0);
   672→   672→    }
   673→   673→    // At the very last chunk: no-op (Finish is an explicit button in the footer).
   674→   674→  }, [
   675→   675→    currentChapter,
   676→   676→    activeChunkIndex,
   677→   677→    currentChapterIndex,
   678→   678→    chapters.length,
   679→   679→  ]);
   680→   680→
   681→   681→  const prevChunk = useCallback(() => {
   682→   682→    if (isFirstChunk) return;
   683→   683→    if (activeChunkIndex > 0) {
   684→   684→      setActiveChunkIndex((i) => i - 1);
   685→   685→    } else if (currentChapterIndex > 0) {
   686→   686→      const prev = chapters[currentChapterIndex - 1];
   687→   687→      setCurrentChapterIndex((i) => i - 1);
   688→   688→      setActiveChunkIndex(Math.max(0, chapterChunkCount(prev) - 1));
   689→   689→    }
   690→   690→  }, [activeChunkIndex, currentChapterIndex, chapters, isFirstChunk]);
   691→   691→
   692→   692→  const jumpToChapter = useCallback((idx: number) => {
   693→   693→    setCurrentChapterIndex(idx);
   694→   694→    setActiveChunkIndex(0);
   695→   695→    setChapterSheetOpen(false);
   696→   696→  }, []);
   697→   697→
   698→   698→  const nextChapter = useCallback(() => {
   699→   699→    if (currentChapterIndex < chapters.length - 1) {
   700→   700→      setCurrentChapterIndex((i) => i + 1);
   701→   701→      setActiveChunkIndex(0);
   702→   702→    }
   703→   703→  }, [currentChapterIndex, chapters.length]);
   704→   704→
   705→   705→  const prevChapter = useCallback(() => {
   706→   706→    if (currentChapterIndex > 0) {
   707→   707→      setCurrentChapterIndex((i) => i - 1);
   708→   708→      setActiveChunkIndex(0);
   709→   709→    }
   710→   710→  }, [currentChapterIndex]);
   711→   711→
   712→   712→  const finish = useCallback(async () => {
   713→   713→    if (!documentId) return;
   714→   714→    try {
   715→   715→      const updated = await patchDocument(documentId, {
   716→   716→        readingProgress: 1,
   717→   717→        lastChunkIndex: globalChunkIndex,
   718→   718→        lastReadAt: new Date().toISOString(),
   719→   719→      });
   720→   720→      setDoc(updated);
   721→   721→      toast.success("🎉 Finished!");
   722→   722→    } catch {
   723→   723→      // ignore
   724→   724→    }
   725→   725→  }, [documentId, globalChunkIndex, setDoc]);
   726→   726→
   727→   727→  /* ---------- keyboard navigation ---------- */
   728→   728→  useEffect(() => {
   729→   729→    const onKey = (e: KeyboardEvent) => {
   730→   730→      const target = e.target as HTMLElement | null;
   731→   731→      const isTyping =
   732→   732→        !!target &&
   733→   733→        (target.tagName === "INPUT" ||
   734→   734→          target.tagName === "TEXTAREA" ||
   735→   735→          target.isContentEditable);
   736→   736→
   737→   737→      if (e.key === "ArrowRight" || (e.key === " " && !isTyping)) {
   738→   738→        e.preventDefault();
   739→   739→        nextChunk();
   740→   740→      } else if (e.key === "ArrowLeft") {
   741→   741→        e.preventDefault();
   742→   742→        prevChunk();
   743→   743→      } else if ((e.key === "f" || e.key === "F") && !isTyping) {
   744→   744→        e.preventDefault();
   745→   745→        update({ focusMode: !settings.focusMode });
   746→   746→      } else if (e.key === "Escape") {
   747→   747→        setChapterSheetOpen(false);
   748→   748→        setSettingsSheetOpen(false);
   749→   749→        setAiSheetOpen(false);
   750→   750→      }
   751→   751→    };
   752→   752→    window.addEventListener("keydown", onKey);
   753→   753→    return () => window.removeEventListener("keydown", onKey);
   754→   754→  }, [nextChunk, prevChunk, settings.focusMode, update]);
   755→   755→
   756→   756→  /* ---------- open Luma panel ---------- */
   757→   757→  const openAiPanel = useCallback(() => {
   758→   758→    setAiSheetOpen(true);
   759→   759→  }, []);
   760→   760→
   761→   761→  /* ---------- scenes overlay: fetch + toggle ---------- */
   762→   762→  const toggleScenesView = useCallback(async () => {
   763→   763→    if (!documentId) return;
   764→   764→    // If turning on and we have no data yet, fetch first.
   765→   765→    if (!scenesView && !scenesData && !scenesLoading) {
   766→   766→      setScenesLoading(true);
   767→   767→      try {
   768→   768→        const { scenes } = await fetchScenes(documentId, false);
   769→   769→        setScenesData(scenes);
   770→   770→      } catch {
   771→   771→        toast.error("Couldn't load scenes. Try asking Luma to cinematize.");
   772→   772→      } finally {
   773→   773→        setScenesLoading(false);
   774→   774→      }
   775→   775→    }
   776→   776→    setScenesView((v) => !v);
   777→   777→  }, [documentId, scenesView, scenesData, scenesLoading]);
   778→   778→
   779→   779→  const regenerateScenes = useCallback(async () => {
   780→   780→    if (!documentId) return;
   781→   781→    setScenesLoading(true);
   782→   782→    try {
   783→   783→      const { scenes } = await fetchScenes(documentId, true);
   784→   784→      setScenesData(scenes);
   785→   785→      toast.success("Scenes regenerated");
   786→   786→    } catch {
   787→   787→      toast.error("Couldn't regenerate scenes.");
   788→   788→    } finally {
   789→   789→      setScenesLoading(false);
   790→   790→    }
   791→   791→  }, [documentId]);
   792→   792→
   793→   793→  const toggleBookmark = useCallback(() => {
   794→   794→    if (!currentChapter) return;
   795→   795→    setBookmarks((prev) => {
   796→   796→      const next = new Set(prev);
   797→   797→      if (next.has(bookmarkKey)) {
   798→   798→        next.delete(bookmarkKey);
   799→   799→        toast("Bookmark removed");
   800→   800→      } else {
   801→   801→        next.add(bookmarkKey);
   802→   802→        toast.success("Bookmarked this chunk");
   803→   803→      }
   804→   804→      return next;
   805→   805→    });
   806→   806→  }, [bookmarkKey, currentChapter]);
   807→   807→
   808→   808→  /* ---------- early returns ---------- */
   809→   809→  if (!documentId) {
   810→   810→    return (
   811→   811→      <div className="flex min-h-dvh flex-col items-center justify-center gap-4 bg-background px-6 text-center text-foreground">
   812→   812→        <BookOpen className="h-8 w-8 text-muted-foreground" aria-hidden />
   813→   813→        <p className="text-lg font-medium">No document selected</p>
   814→   814→        <Button onClick={() => go("library")}>Open library</Button>
   815→   815→      </div>
   816→   816→    );
   817→   817→  }
   818→   818→
   819→   819→  if (loading) {
   820→   820→    return <LoadingScreen label="Opening document" className="min-h-dvh" />;
   821→   821→  }
   822→   822→
   823→   823→  if (error) {
   824→   824→    return (
   825→   825→      <div className="flex min-h-dvh flex-col items-center justify-center gap-4 bg-background px-6 text-center text-foreground">
   826→   826→        <div className="max-w-md rounded-lg border border-border bg-card p-6 text-left">
   827→   827→          <p className="text-lg font-medium">Couldn&apos;t open this document</p>
   828→   828→          <p className="mt-2 text-sm text-muted-foreground">{error}</p>
   829→   829→          <Button className="mt-4" onClick={() => go("library")}>
   830→   830→            Back to library
   831→   831→          </Button>
   832→   832→        </div>
   833→   833→      </div>
   834→   834→    );
   835→   835→  }
   836→   836→
   837→   837→  if (!doc) return null;
   838→   838→
   839→   839→  if (doc.status === "processing") {
   840→   840→    return (
   841→   841→      <div className="flex min-h-dvh flex-col items-center justify-center gap-4 bg-background px-6 text-center text-foreground">
   842→   842→        <LemniscateSpinner size={72} />
   843→   843→        <p className="text-lg font-medium">Still processing</p>
   844→   844→        <p className="text-sm text-muted-foreground">
   845→   845→          Check back in a moment — we&apos;re parsing this file.
   846→   846→        </p>
   847→   847→        <Button variant="outline" onClick={() => go("library")}>
   848→   848→          Back to library
   849→   849→        </Button>
   850→   850→      </div>
   851→   851→    );
   852→   852→  }
   853→   853→
   854→   854→  if (doc.status === "error" || chapters.length === 0) {
   855→   855→    return (
   856→   856→      <div className="flex min-h-dvh flex-col items-center justify-center gap-4 bg-background px-6 text-center text-foreground">
   857→   857→        <div className="max-w-md rounded-lg border border-border bg-card p-6 text-left">
   858→   858→          <p className="text-lg font-medium">
   859→   859→            {doc.status === "error"
   860→   860→              ? "This document failed to parse"
   861→   861→              : "No readable content yet"}
   862→   862→          </p>
   863→   863→          <p className="mt-2 text-sm text-muted-foreground">
   864→   864→            {doc.error ?? "Try re-uploading the file."}
   865→   865→          </p>
   866→   866→          <Button className="mt-4" onClick={() => go("library")}>
   867→   867→            Back to library
   868→   868→          </Button>
   869→   869→        </div>
   870→   870→      </div>
   871→   871→    );
   872→   872→  }
   873→   873→
   874→   874→  /* ---------- main render ---------- */
   875→   875→
   876→   876→  const sourceLabel = SOURCE_LABELS[doc.sourceType] ?? "DOC";
   877→   877→
   878→   878→  // Resolve the currently-relevant summary view-model.
   879→   879→
   880→   880→  return (
   881→   881→    <div className="min-h-dvh bg-background text-foreground">
   882→   882→      {/* ---- ReaderHeader ---- */}
   883→   883→      <header className="sticky top-0 z-30 border-b border-border bg-background/80 backdrop-blur-md">
   884→   884→        <div className="mx-auto flex max-w-5xl items-center justify-between gap-3 px-4 py-2.5 sm:px-6">

   886→   886→          <div className="flex min-w-0 items-center gap-2">

   889→   889→              size="sm"
