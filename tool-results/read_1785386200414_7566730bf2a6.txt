    45→    45→function scrollToAnchor(e: React.MouseEvent, href: string) {
    46→    46→  e.preventDefault();
    47→    47→  if (typeof document === "undefined") return;
    48→    48→  // Guard against invalid selectors: "#", empty strings, or malformed hrefs.
    49→    49→  // querySelector throws a SyntaxError for "#" alone.
    50→    50→  if (!href || href === "#" || href.length < 2) return;
    51→    51→  try {
    52→    52→    const el = document.querySelector(href);
    53→    53→    if (el) el.scrollIntoView({ behavior: "smooth", block: "start" });
    54→    54→  } catch {
    55→    55→    // Invalid selector — silently ignore rather than crash.
    56→    56→  }
    57→    57→}
