   425→/**
   426→ * Effective chunk count for a chapter. If the chapter has been OCR-refined,
   427→ * we split the refined text into ~1200-char chunks at paragraph boundaries.
   428→ * Otherwise we use the parser-produced chunks.
   429→ */
   430→function chapterChunkCount(ch: Chapter | undefined): number {
   431→  if (!ch) return 0;
   432→  if (ch.refinedText) {
   433→    // Split refined text into chunks for focus-mode navigation
   434→    const paras = ch.refinedText.split(/\n{2,}/).filter(Boolean);
   435→    return Math.max(1, Math.ceil(paras.length / 3));
   436→  }
   437→  return ch.chunks.length;
   438→}
   439→
   440→/** Split refined text into paragraph groups for rendering. */
   441→function refinedTextGroups(text: string): string[] {
   442→  const paras = text.split(/\n{2,}/).map((p) => p.trim()).filter(Boolean);
   443→  const groups: string[] = [];
   444→  for (let i = 0; i < paras.length; i += 3) {
   445→    groups.push(paras.slice(i, i + 3).join("\n\n"));
   446→  }
   447→  return groups;
   448→}

   452→export default function ReaderView() {
   453→  const documentId = useNav((s) => s.activeDocumentId);
   454→  const go = useNav((s) => s.go);
   455→
   456→  const { doc, content: docContent, loading, error, setDoc, setContent, refresh } = useDocument(documentId);
   457→  const { settings, update } = useReaderSettings();
   458→
   459→  const [currentChapterIndex, setCurrentChapterIndex] = useState(0);
   460→  const [activeChunkIndex, setActiveChunkIndex] = useState(0);
   461→
   462→  const [chapterSheetOpen, setChapterSheetOpen] = useState(false);
   463→  const [settingsSheetOpen, setSettingsSheetOpen] = useState(false);
   464→  const [aiSheetOpen, setAiSheetOpen] = useState(false);
   465→  const [scenesView, setScenesView] = useState(false);
   466→  const [scenesData, setScenesData] = useState<AiScene[] | null>(null);
   467→  const [scenesLoading, setScenesLoading] = useState(false);
   468→
   469→  const [bookmarks, setBookmarks] = useState<Set<string>>(new Set());
   470→  const [sessionMarkedRead, setSessionMarkedRead] = useState(false);
   471→
   472→  // --- Per-chapter OCR refinement state ---
   473→  const [refining, setRefining] = useState(false);
   474→  const [refineTriggered, setRefineTriggered] = useState<Set<number>>(new Set());
   475→  const refineAbortRef = useRef<number | null>(null);
   476→
   477→  const content = docContent;
   478→  const chapters: Chapter[] = content?.chapters ?? [];
   479→
   480→  const totalChunks = useMemo(
   481→    () => chapters.reduce((acc, ch) => acc + chapterChunkCount(ch), 0),
   482→    [chapters],
   483→  );
   484→
   485→  const globalChunkIndex = useMemo(() => {
   486→    let g = 0;
   487→    for (let i = 0; i < currentChapterIndex && i < chapters.length; i++) {
   488→      g += chapterChunkCount(chapters[i]);
   489→    }
   490→    g += activeChunkIndex;
   491→    return g;
   492→  }, [chapters, currentChapterIndex, activeChunkIndex]);
   493→
   494→  const readingProgress =
   495→    totalChunks > 0 ? Math.min(1, (globalChunkIndex + 1) / totalChunks) : 0;
   496→  const progressPct = Math.round(readingProgress * 100);
   497→
   498→  const currentChapter = chapters[currentChapterIndex];
   499→  const currentChunk = currentChapter?.chunks[activeChunkIndex];
   500→
   501→  const isLastChunk =
   502→    chapters.length > 0 &&
   503→    currentChapterIndex === chapters.length - 1 &&
   504→    activeChunkIndex === chapterChunkCount(currentChapter) - 1;
   505→  const isFirstChunk = currentChapterIndex === 0 && activeChunkIndex === 0;
   506→
   507→  const bookmarkKey = `${currentChapterIndex}:${activeChunkIndex}`;
   508→  const isBookmarked = bookmarks.has(bookmarkKey);
   509→
   510→  /* ---------- initialize position from doc.lastChunkIndex ---------- */
   511→  useEffect(() => {
   512→    if (!doc || chapters.length === 0) return;
   513→    const lastGlobal = Math.max(
   514→      0,
   515→      Math.min(doc.lastChunkIndex ?? 0, totalChunks - 1),
   516→    );
   517→    let chapterIdx = 0;
   518→    let chunkIdx = 0;
   519→    let acc = 0;
   520→    let placed = false;
   521→    for (let i = 0; i < chapters.length; i++) {
   522→      const len = chapterChunkCount(chapters[i]);
   523→      if (acc + len > lastGlobal) {
   524→        chapterIdx = i;
   525→        chunkIdx = lastGlobal - acc;
   526→        placed = true;
   527→        break;
   528→      }
   529→      acc += len;
   530→    }
   531→    if (!placed) {
   532→      chapterIdx = chapters.length - 1;
   533→      chunkIdx = chapterChunkCount(chapters[chapters.length - 1]) - 1;
   534→    }
   535→    setCurrentChapterIndex(chapterIdx);
   536→    setActiveChunkIndex(Math.max(0, chunkIdx));
   537→    // Only on doc open — subsequent changes come from navigation.
   538→  }, [doc?.id]);
   539→
   540→  // Clear refining state when chapter changes (the previous chapter's
   541→  // refinement job is abandoned — its result will be ignored via the abort ref)
   542→  useEffect(() => {
   543→    setRefining(false);
   544→    refineAbortRef.current = null;
   545→  }, [currentChapterIndex]);
   546→
   547→  /* ---------- per-chapter OCR refinement (background, non-blocking) ----------
   548→   * When a chapter is opened, it renders instantly with raw text. Then this
   549→   * effect triggers OCR refinement in the background. When refinement
   550→   * completes, the chapter's `refinedText` is updated and the reader
   551→   * re-renders with the cleaned text. The refinement is persisted server-side
   552→   * so the chapter is never reprocessed. */
   553→  const currentChapterRefined = currentChapter?.refinedText;
   554→  useEffect(() => {
   555→    if (!documentId || !content || !currentChapter) return;
   556→    const chIdx = currentChapterIndex;
   557→
   558→    // Already refined? Skip.
   559→    if (currentChapterRefined) return;
   560→    // Already triggered for this chapter? Skip.
   561→    if (refineTriggered.has(chIdx)) return;
   562→
   563→    // Mark as triggered
   564→    setRefineTriggered((prev) => new Set(prev).add(chIdx));
   565→    setRefining(true);
   566→    refineAbortRef.current = chIdx;
   567→
   568→    let cancelled = false;
   569→    refineChapter(documentId, chIdx, false)
   570→      .then((result) => {
   571→        if (cancelled || refineAbortRef.current !== chIdx) return;
   572→        if (result.refined && result.refinedText) {
   573→          // Update the chapter's refinedText in local state
   574→          setContent((prev) => {
   575→            if (!prev) return prev;
   576→            const newChapters = prev.chapters.map((ch, i) =>
   577→              i === chIdx ? { ...ch, refinedText: result.refinedText } : ch,
   578→            );
   579→            return { ...prev, chapters: newChapters };
   580→          });
   581→        }
   582→      })
   583→      .catch(() => {
   584→        // Best-effort — silently ignore (rate limits, network errors, etc.)
   585→      })
   586→      .finally(() => {
   587→        if (cancelled || refineAbortRef.current !== chIdx) return;
   588→        setRefining(false);
   589→      });
   590→
   591→    // Safety timeout — don't show the indicator for more than 60 seconds
   592→    const safetyTimer = setTimeout(() => {
   593→      if (!cancelled && refineAbortRef.current === chIdx) {
   594→        setRefining(false);
   595→      }
   596→    }, 60_000);
   597→
   598→    return () => {
   599→      cancelled = true;
   600→      clearTimeout(safetyTimer);
   601→    };
   602→  }, [documentId, currentChapterIndex, currentChapterRefined, content, refineTriggered]);
   603→
   604→  /* ---------- scroll active chunk into view ---------- */
   605→  const scrollRafRef = useRef<number | null>(null);
   606→  useEffect(() => {
   607→    if (scrollRafRef.current) cancelAnimationFrame(scrollRafRef.current);
   608→    scrollRafRef.current = requestAnimationFrame(() => {
   609→      if (!currentChapter) return;
   610→      const el = document.querySelector(
   611→        `.reader-article [data-ordinal="${activeChunkIndex}"]`,
   612→      );
   613→      if (el) {
   614→        el.scrollIntoView({ behavior: "smooth", block: "center" });
   615→      }
   616→    });
   617→    return () => {
   618→      if (scrollRafRef.current) cancelAnimationFrame(scrollRafRef.current);

   639→        readingProgress,
   640→        lastChunkIndex: globalChunkIndex,
   641→      };
   642→      if (!sessionMarkedReadRef.current) {
   643→        patch.lastReadAt = new Date().toISOString();

   645→      }
   646→      try {
   647→        const updated = await patchDocument(documentId, patch);

   650→        // ignore network hiccups
   651→      }
   652→    }, 600);
   653→    return () => {
   654→      if (patchTimer.current) clearTimeout(patchTimer.current);

   658→    readingProgress,
   659→    documentId,
