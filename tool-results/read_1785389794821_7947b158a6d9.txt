    69→  recent: "Most recent",
    70→  title: "Title (A–Z)",
    71→  size: "File size",
    72→  progress: "Reading progress",

    75→const STATUS_OPTIONS: { value: StatusFilter; label: string }[] = [
    76→  { value: "all", label: "All" },

    78→  { value: "processing", label: "Processing" },

    90→// ─── Page ───────────────────────────────────────────────────────────────
    91→
    92→export default function LibraryView() {
    93→  const { go } = useNav();
    94→  const { docs, loading, error, refresh, setDocs } = useDocuments();
    95→  const prefs = useLibraryPrefs();
    96→
    97→  const [query, setQuery] = useState("");
    98→  const [dragging, setDragging] = useState(false);
    99→  const [deleteTarget, setDeleteTarget] = useState<DocumentRow | null>(null);
   100→  const [deleting, setDeleting] = useState(false);
   101→
   102→  // Derived: all source types present in the library (for filter chips)
   103→  const availableSourceTypes = useMemo<SourceType[]>(() => {
   104→    const set = new Set<SourceType>();
   105→    docs.forEach((d) => set.add(d.sourceType));
   106→    return Array.from(set).sort();
   107→  }, [docs]);
   108→
   109→  const hasActiveFilters =
   110→    prefs.status !== "all" ||
   111→    prefs.showFavoritesOnly ||
   112→    prefs.showRecentOnly ||
   113→    prefs.sourceTypes.length > 0 ||
   114→    query.trim().length > 0;
   115→
   116→  // Filter + sort pipeline
   117→  const filtered = useMemo<DocumentRow[]>(() => {
   118→    let list = docs.slice();
   119→    if (prefs.status !== "all") {
   120→      list = list.filter((d) => d.status === prefs.status);
   121→    }
   122→    if (prefs.showFavoritesOnly) {
   123→      list = list.filter((d) => d.favorite);
   124→    }
   125→    if (prefs.showRecentOnly) {
   126→      list = list.filter((d) => isRecent(d.createdAt));
   127→    }
   128→    if (prefs.sourceTypes.length > 0) {
   129→      const s = new Set<SourceType>(prefs.sourceTypes as SourceType[]);
   130→      list = list.filter((d) => s.has(d.sourceType));
   131→    }
   132→    const q = query.trim().toLowerCase();
   133→    if (q) {
   134→      list = list.filter(
   135→        (d) =>
   136→          d.title.toLowerCase().includes(q) ||
   137→          (d.author ?? "").toLowerCase().includes(q) ||
   138→          d.tags.some((t) => t.toLowerCase().includes(q)),
   139→      );
   140→    }
   141→    switch (prefs.sort) {
   142→      case "title":
   143→        list.sort((a, b) => a.title.localeCompare(b.title));
   144→        break;
   145→      case "size":
   146→        list.sort((a, b) => b.byteSize - a.byteSize);
   147→        break;
   148→      case "progress":
   149→        list.sort((a, b) => b.readingProgress - a.readingProgress);
   150→        break;
   151→      default:
   152→        list.sort(
   153→          (a, b) =>
   154→            new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime(),
   155→        );
   156→    }
   157→    return list;
   158→  }, [docs, prefs.status, prefs.showFavoritesOnly, prefs.showRecentOnly, prefs.sourceTypes, prefs.sort, query]);
   159→
   160→  // ── Drag-and-drop file import ──────────────────────────────────────────
   161→  async function handleFiles(files: FileList | File[]): Promise<void> {
   162→    const arr = Array.from(files);
   163→    if (arr.length === 0) return;
   164→    for (const file of arr) {
   165→      toast(`Importing ${file.name}…`);
   166→      try {
   167→        const { document, error: err } = await uploadFile(file);
   168→        if (err || !document) {
   169→          toast.error(err ?? `Failed to import ${file.name}`);
   170→        } else {
   171→          toast.success(`Imported "${document.title}"`);
   172→        }
   173→      } catch {
   174→        toast.error(`Failed to import ${file.name}`);
   175→      }
   176→    }
   177→    await refresh();
   178→  }
   179→
   180→  function onDragOver(e: DragEvent<HTMLDivElement>): void {
   181→    if (Array.from(e.dataTransfer.types).includes("Files")) {
   182→      e.preventDefault();
   183→      e.dataTransfer.dropEffect = "copy";
   184→      if (!dragging) setDragging(true);
   185→    }
   186→  }
   187→  function onDragLeave(e: DragEvent<HTMLDivElement>): void {
   188→    // Only clear when leaving the outer container (not bubbling children)
   189→    if (e.currentTarget === e.target) {
   190→      setDragging(false);
   191→    }
   192→  }
   193→  function onDrop(e: DragEvent<HTMLDivElement>): void {
   194→    e.preventDefault();
   195→    setDragging(false);
   196→    if (e.dataTransfer.files && e.dataTransfer.files.length > 0) {
   197→      void handleFiles(e.dataTransfer.files);
   198→    }
   199→  }
   200→
   201→  // ── Favorite toggle (optimistic) ───────────────────────────────────────
   202→  async function toggleFavorite(doc: DocumentRow): Promise<void> {
   203→    const next = !doc.favorite;
   204→    setDocs((prev) =>
   205→      prev.map((d) => (d.id === doc.id ? { ...d, favorite: next } : d)),
   206→    );
   207→    try {
   208→      await patchDocument(doc.id, { favorite: next });
   209→      toast(next ? "Added to favorites" : "Removed from favorites");
   210→    } catch {
   211→      setDocs((prev) =>
   212→        prev.map((d) => (d.id === doc.id ? { ...d, favorite: doc.favorite } : d)),
   213→      );
   214→      toast.error("Failed to update favorite");
   215→    }
   216→  }
   217→
   218→  // ── Delete flow ────────────────────────────────────────────────────────
   219→  async function confirmDelete(): Promise<void> {
   220→    if (!deleteTarget) return;
   221→    setDeleting(true);
   222→    try {
   223→      await deleteDocument(deleteTarget.id);
   224→      setDocs((prev) => prev.filter((d) => d.id !== deleteTarget.id));
   225→      toast.success("Document deleted");
   226→      setDeleteTarget(null);
   227→    } catch {
   228→      toast.error("Failed to delete document");
   229→    } finally {
   230→      setDeleting(false);
   231→    }
   232→  }
   233→
   234→  function clearFilters(): void {
   235→    setLibraryPrefs({
   236→      status: "all",
   237→      sourceTypes: [],
   238→      showFavoritesOnly: false,
   239→      showRecentOnly: false,
   240→    });
   241→    setQuery("");
   242→  }
   243→
   244→  return (
   245→    <div
   246→      className="dashboard-noir min-h-dvh"
   247→      onDragOver={onDragOver}
   248→      onDragLeave={onDragLeave}
   249→      onDrop={onDrop}
   250→    >
   251→      <AppHeader
   252→        breadcrumbs={[
   253→          { label: "Dashboard", onClick: () => go("dashboard") },
   254→          { label: "Library" },
   255→        ]}
   256→      />
   257→
   258→      <main
   259→        id="main-content"
   260→        className="mx-auto max-w-7xl space-y-6 px-4 py-8 sm:px-6 lg:px-8"
   261→      >
   262→        {/* Header */}
   263→        <header className="flex flex-col items-start justify-between gap-4 sm:flex-row sm:items-end">
   264→          <div className="min-w-0 space-y-2">
   265→            <p className="noir-eyebrow">Your collection</p>
   266→            <h1 className="noir-display text-4xl">Library</h1>
   267→            <p className="text-sm text-muted-foreground">
   268→              Organize, filter, and read documents stored on this device.
   269→            </p>
   270→          </div>
   271→          <Button
   272→            type="button"
   273→            onClick={() => go("upload")}
   274→            className="noir-btn-gold shrink-0 rounded-full px-4 text-sm font-semibold"
   275→          >
   276→            <UploadIcon className="size-4" />
   277→            Import
   278→          </Button>
   279→        </header>
   280→
   281→        {/* Toolbar */}
   282→        <Toolbar
   283→          query={query}
   284→          onQuery={setQuery}
   285→          availableSourceTypes={availableSourceTypes}
   286→          hasActiveFilters={hasActiveFilters}
   287→          onClearFilters={clearFilters}
   288→        />
   289→
   290→        {/* Result count line */}
   291→        {!loading && docs.length > 0 ? (
   292→          <div className="flex items-center justify-between px-1 text-xs text-muted-foreground">
   293→            <span>
   294→              Showing <span className="tabular-nums">{filtered.length}</span> of{" "}
   295→              <span className="tabular-nums">{docs.length}</span>{" "}
   296→              {docs.length === 1 ? "document" : "documents"}
   297→            </span>
   298→            {hasActiveFilters ? (
   299→              <button
   300→                type="button"
   301→                onClick={clearFilters}
   302→                className="inline-flex items-center gap-1 rounded-full px-2 py-1 transition hover:text-foreground"
   303→              >
   304→                <X className="size-3" />
   305→                Clear filters
   306→              </button>
   307→            ) : null}
   308→          </div>
   309→        ) : null}
   310→
   311→        {/* Grid / List / Empty states */}
   312→        {loading ? (
   313→          <LoadingState />
   314→        ) : error ? (
   315→          <ErrorState message={error} onRetry={() => void refresh()} />
   316→        ) : docs.length === 0 ? (
   317→          <EmptyLibraryState onImport={() => go("upload")} />
   318→        ) : filtered.length === 0 ? (
   319→          <NoMatchesState onClear={clearFilters} />
   320→        ) : prefs.view === "grid" ? (
   321→          <DocumentGrid
   322→            docs={filtered}
   323→            onOpen={(id) => go("reader", { documentId: id })}
   324→            onToggleFavorite={toggleFavorite}
   325→            onDeleteRequest={setDeleteTarget}
   326→          />
   327→        ) : (
   328→          <DocumentList
   329→            docs={filtered}
   330→            onOpen={(id) => go("reader", { documentId: id })}
   331→            onToggleFavorite={toggleFavorite}
   332→            onDeleteRequest={setDeleteTarget}
   333→          />
   334→        )}
   335→      </main>
   336→
   337→      {/* Drag-and-drop overlay */}
   338→      {dragging ? (
   339→        <div
   340→          className="pointer-events-none fixed inset-0 z-50 flex items-center justify-center"
   341→          aria-hidden
   342→        >
   343→          <div className="absolute inset-0 bg-black/60 backdrop-blur-sm" />
   344→          <div className="noir-card relative flex flex-col items-center gap-3 px-12 py-10">
   345→            <div className="grid size-14 place-items-center rounded-full bg-[color-mix(in_oklab,var(--noir-gold)_18%,transparent)]">
   346→              <UploadIcon className="size-7 text-[var(--noir-gold)]" />
   347→            </div>
   348→            <p className="noir-display text-2xl">Drop to import</p>
   349→            <p className="text-xs text-muted-foreground">
   350→              Release to add files to your library
   351→            </p>
   352→          </div>
   353→        </div>
   354→      ) : null}
   355→
   356→      {/* Delete confirmation */}
   357→      <AlertDialog
   358→        open={!!deleteTarget}
   359→        onOpenChange={(o) => !o && setDeleteTarget(null)}
   360→      >
   361→        <AlertDialogContent>
   362→          <AlertDialogHeader>
   363→            <AlertDialogTitle>Delete document?</AlertDialogTitle>
   364→            <AlertDialogDescription>
   365→              {deleteTarget
   366→                ? `"${deleteTarget.title}" will be permanently removed from your library. This action cannot be undone.`
   367→                : ""}
   368→            </AlertDialogDescription>
   369→          </AlertDialogHeader>
   370→          <AlertDialogFooter>
   371→            <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>
   372→            <AlertDialogAction
   373→              onClick={(e) => {
   374→                e.preventDefault();
   375→                void confirmDelete();
   376→              }}
   377→              disabled={deleting}
   378→              className="bg-destructive text-white hover:bg-destructive/90"
   379→            >
   380→              {deleting ? (
   381→                <>
   382→                  <Loader2 className="size-4 animate-spin" />
   383→                  Deleting…
   384→                </>
   385→              ) : (
   386→                "Delete"
   387→              )}
   388→            </AlertDialogAction>
   389→          </AlertDialogFooter>
   390→        </AlertDialogContent>
   391→      </AlertDialog>
   392→    </div>
   393→  );
   394→}

   398→function Toolbar({
   399→  query,
   400→  onQuery,
   401→  availableSourceTypes,
   402→  hasActiveFilters,
   403→  onClearFilters,
   404→}: {
   405→  query: string;
   406→  onQuery: (v: string) => void;
   407→  availableSourceTypes: SourceType[];
   408→  hasActiveFilters: boolean;
   409→  onClearFilters: () => void;
   410→}): ReactNode {
   411→  const prefs = useLibraryPrefs();
   412→
   413→  return (
   414→    <div className="noir-card space-y-3 p-3">
   415→      <div className="grid grid-cols-1 items-center gap-3 md:grid-cols-[minmax(0,1fr)_auto_auto_auto]">
   416→        {/* Search */}
   417→        <div className="relative">
   418→          <Search className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
   419→          <Input
   420→            type="search"
   421→            value={query}
   422→            onChange={(e) => onQuery(e.target.value)}
   423→            placeholder="Search by title, author, tag…"
   424→            className="pl-9 pr-9"
   425→            aria-label="Search library"
   426→          />
   427→          {query ? (
   428→            <button
   429→              type="button"
   430→              onClick={() => onQuery("")}
   431→              aria-label="Clear search"
   432→              className="absolute right-2 top-1/2 -translate-y-1/2 rounded-md p-1 text-muted-foreground transition hover:text-foreground"
   433→            >
   434→              <X className="size-3.5" />
   435→            </button>
   436→          ) : null}
   437→        </div>
   438→
   439→        {/* Status filter chips */}
   440→        <div
   441→          className="flex items-center gap-1"
   442→          role="group"
   443→          aria-label="Filter by status"
   444→        >
   445→          {STATUS_OPTIONS.map((s) => {
   446→            const active = prefs.status === s.value;
   447→            return (
   448→              <button
   449→                key={s.value}
   450→                type="button"
   451→                onClick={() => setLibraryPrefs({ status: s.value })}
   452→                aria-pressed={active}
   453→                className={cn("noir-chip cursor-pointer transition", active && "noir-chip-gold")}
   454→              >
   455→                {s.label}
   456→              </button>
   457→            );
   458→          })}
   459→        </div>
   460→
   461→        {/* Sort select */}
   462→        <div className="relative">
   463→          <label htmlFor="library-sort" className="sr-only">
   464→            Sort
   465→          </label>
   466→          <select
   467→            id="library-sort"
   468→            value={prefs.sort}
   469→            onChange={(e) => setLibraryPrefs({ sort: e.target.value as SortKey })}
   470→            className="noir-btn-ghost inline-flex h-9 cursor-pointer appearance-none items-center gap-2 rounded-lg border border-border bg-transparent pl-3 pr-8 text-sm text-foreground transition hover:text-foreground"
   471→          >
   472→            {(Object.keys(SORT_LABELS) as SortKey[]).map((k) => (
   473→              <option key={k} value={k} className="bg-background text-foreground">
   474→                {SORT_LABELS[k]}
   475→              </option>
   476→            ))}
   477→          </select>
   478→          <ChevronDown className="pointer-events-none absolute right-2 top-1/2 size-3.5 -translate-y-1/2 text-muted-foreground" />
   479→        </div>
   480→
   481→        {/* View toggle */}
   482→        <div
   483→          className="inline-flex items-center gap-0.5 rounded-lg border border-border p-0.5"
   484→          role="tablist"
   485→          aria-label="View mode"
   486→        >
   487→          <button
   488→            type="button"
   489→            role="tab"
   490→            aria-selected={prefs.view === "grid"}
   491→            onClick={() => setLibraryPrefs({ view: "grid" })}
   492→            className={cn(
   493→              "grid size-8 place-items-center rounded-md transition",
   494→              prefs.view === "grid"
   495→                ? "bg-[color-mix(in_oklab,var(--noir-gold)_18%,transparent)] text-[var(--noir-gold-soft)]"
   496→                : "text-muted-foreground hover:text-foreground",
   497→            )}
   498→            aria-label="Grid view"
   499→          >
   500→            <Grid3x3 className="size-4" />
   501→          </button>
   502→          <button
   503→            type="button"
   504→            role="tab"
   505→            aria-selected={prefs.view === "list"}
   506→            onClick={() => setLibraryPrefs({ view: "list" })}
   507→            className={cn(
   508→              "grid size-8 place-items-center rounded-md transition",
   509→              prefs.view === "list"
   510→                ? "bg-[color-mix(in_oklab,var(--noir-gold)_18%,transparent)] text-[var(--noir-gold-soft)]"
   511→                : "text-muted-foreground hover:text-foreground",
   512→            )}
   513→            aria-label="List view"
   514→          >
   515→            <LayoutList className="size-4" />
   516→          </button>
   517→        </div>
   518→      </div>
   519→
   520→      {/* Second row: favorites/recent toggles + source-type chips */}
   521→      <div className="flex flex-wrap items-center gap-2 border-t border-border pt-3">
   522→        <button
   523→          type="button"
   524→          onClick={() =>
   525→            setLibraryPrefs({ showFavoritesOnly: !prefs.showFavoritesOnly })
   526→          }
   527→          aria-pressed={prefs.showFavoritesOnly}
   528→          className={cn(
   529→            "noir-chip cursor-pointer transition",
   530→            prefs.showFavoritesOnly && "noir-chip-gold",
   531→          )}
   532→        >
   533→          <Heart
   534→            className="size-3"
   535→            fill={prefs.showFavoritesOnly ? "currentColor" : "none"}
   536→          />
   537→          Favorites
   538→        </button>
   539→        <button
   540→          type="button"
   541→          onClick={() =>
   542→            setLibraryPrefs({ showRecentOnly: !prefs.showRecentOnly })
   543→          }
   544→          aria-pressed={prefs.showRecentOnly}
   545→          className={cn(
   546→            "noir-chip cursor-pointer transition",
   547→            prefs.showRecentOnly && "noir-chip-gold",
   548→          )}
   549→        >
   550→          <Clock className="size-3" />
   551→          Recent
   552→        </button>
   553→
   554→        {availableSourceTypes.length > 0 ? (
   555→          <>
   556→            <span className="mx-1 h-4 w-px bg-border" aria-hidden />
   557→            <span className="text-[11px] uppercase tracking-wider text-muted-foreground">
   558→              Type
   559→            </span>
   560→            {availableSourceTypes.map((t) => {
   561→              const active = prefs.sourceTypes.includes(t);
   562→              return (
   563→                <button
   564→                  key={t}
   565→                  type="button"
   566→                  onClick={() =>
   567→                    setLibraryPrefs({
   568→                      sourceTypes: active
   569→                        ? prefs.sourceTypes.filter((x) => x !== t)
   570→                        : [...prefs.sourceTypes, t],
   571→                    })
   572→                  }
   573→                  aria-pressed={active}
   574→                  className={cn(
   575→                    "noir-chip cursor-pointer transition",
   576→                    active && "noir-chip-gold",
   577→                  )}
   578→                >
   579→                  {sourceLabel(t)}
   580→                </button>
   581→              );
   582→            })}
   583→          </>
   584→        ) : null}
   585→
   586→        {hasActiveFilters ? (
   587→          <button
   588→            type="button"
   589→            onClick={onClearFilters}
   590→            className="ml-auto inline-flex items-center gap-1 rounded-full px-2 py-1 text-[11px] text-muted-foreground transition hover:text-foreground"
   591→          >
   592→            <X className="size-3" />
   593→            Reset
   594→          </button>
   595→        ) : null}
   596→      </div>
   597→    </div>
   598→  );
   599→}

   601→// ─── Grid view ──────────────────────────────────────────────────────────
   602→
   603→function DocumentGrid({
   604→  docs,
   605→  onOpen,
   606→  onToggleFavorite,
   607→  onDeleteRequest,
