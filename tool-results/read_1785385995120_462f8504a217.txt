   375→   375→/**
   376→   376→ * Effective chunk count for a chapter. If the chapter has been OCR-refined,
   377→   377→ * we split the refined text into ~1200-char chunks at paragraph boundaries.
   378→   378→ * Otherwise we use the parser-produced chunks.
   379→   379→ */
   380→   380→function chapterChunkCount(ch: Chapter | undefined): number {
   381→   381→  if (!ch) return 0;
   382→   382→  if (ch.refinedText) {
   383→   383→    // Split refined text into chunks for focus-mode navigation
   384→   384→    const paras = ch.refinedText.split(/\n{2,}/).filter(Boolean);
   385→   385→    return Math.max(1, Math.ceil(paras.length / 3));
   386→   386→  }
   387→   387→  return ch.chunks.length;
   388→   388→}
   389→   389→
   390→   390→/** Split refined text into paragraph groups for rendering. */
   391→   391→function refinedTextGroups(text: string): string[] {
   392→   392→  const paras = text.split(/\n{2,}/).map((p) => p.trim()).filter(Boolean);
   393→   393→  const groups: string[] = [];
   394→   394→  for (let i = 0; i < paras.length; i += 3) {
   395→   395→    groups.push(paras.slice(i, i + 3).join("\n\n"));
   396→   396→  }
   397→   397→  return groups;
   398→   398→}

   504→   504→  const currentChunk = currentChapter?.chunks[activeChunkIndex];
   505→   505→
   506→   506→  const isLastChunk =
   507→   507→    chapters.length > 0 &&
   508→   508→    currentChapterIndex === chapters.length - 1 &&
   509→   509→    activeChunkIndex === chapterChunkCount(currentChapter) - 1;

   512→   512→  const bookmarkKey = `${currentChapterIndex}:${activeChunkIndex}`;
   513→   513→  const isBookmarked = bookmarks.has(bookmarkKey);

   519→   519→      0,
   520→   520→      Math.min(doc.lastChunkIndex ?? 0, totalChunks - 1),
   521→   521→    );

   523→   523→    let chunkIdx = 0;

   530→   530→        chunkIdx = lastGlobal - acc;

   532→   532→        break;
   533→   533→      }
   534→   534→      acc += len;
   535→   535→    }
   536→   536→    if (!placed) {

   559→   559→  // empty array if not. We auto-load on first open of the Scenes tab so any
   560→   560→  // previously-generated cinematic scenes show up immediately.

   573→   573→        if (cancelled) return;
   574→   574→        setScenes([]);
   575→   575→        setScenesFetched(true);
   576→   576→      })
   577→   577→      .finally(() => {
   578→   578→        if (!cancelled) setScenesLoading(false);
   579→   579→      });

   581→   581→      cancelled = true;
   582→   582→    };

   584→   584→
   585→   585→  // Clear refining state when chapter changes (the previous chapter's
   586→   586→  // refinement job is abandoned — its result will be ignored via the abort ref)
   587→   587→  useEffect(() => {
   588→   588→    setRefining(false);
   589→   589→    refineAbortRef.current = null;
   590→   590→  }, [currentChapterIndex]);

   592→   592→  /* ---------- per-chapter OCR refinement (background, non-blocking) ----------
   593→   593→   * When a chapter is opened, it renders instantly with raw text. Then this
   594→   594→   * effect triggers OCR refinement in the background. When refinement
   595→   595→   * completes, the chapter's `refinedText` is updated and the reader
   596→   596→   * re-renders with the cleaned text. The refinement is persisted server-side
   597→   597→   * so the chapter is never reprocessed. */
   598→   598→  const currentChapterRefined = currentChapter?.refinedText;
   599→   599→  useEffect(() => {
   600→   600→    if (!documentId || !content || !currentChapter) return;
   601→   601→    const chIdx = currentChapterIndex;
   602→   602→
   603→   603→    // Already refined? Skip.
   604→   604→    if (currentChapterRefined) return;
   605→   605→    // Already triggered for this chapter? Skip.
   606→   606→    if (refineTriggered.has(chIdx)) return;

   609→   609→    setRefineTriggered((prev) => new Set(prev).add(chIdx));
   610→   610→    setRefining(true);
   611→   611→    refineAbortRef.current = chIdx;

   614→   614→    refineChapter(documentId, chIdx, false)
   615→   615→      .then((result) => {
   616→   616→        if (cancelled || refineAbortRef.current !== chIdx) return;

   618→   618→          // Update the chapter's refinedText in local state
   619→   619→          setContent((prev) => {

   622→   622→              i === chIdx ? { ...ch, refinedText: result.refinedText } : ch,
   623→   623→            );
   624→   624→            return { ...prev, chapters: newChapters };
   625→   625→          });
   626→   626→        }
   627→   627→      })
   628→   628→      .catch(() => {
   629→   629→        // Best-effort — silently ignore (rate limits, network errors, etc.)
   630→   630→      })
   631→   631→      .finally(() => {
   632→   632→        if (cancelled || refineAbortRef.current !== chIdx) return;
   633→   633→        setRefining(false);
   634→   634→      });

   636→   636→    // Safety timeout — don't show the indicator for more than 60 seconds
   637→   637→    const safetyTimer = setTimeout(() => {
   638→   638→      if (!cancelled && refineAbortRef.current === chIdx) {
   639→   639→        setRefining(false);
   640→   640→      }
   641→   641→    }, 60_000);
   642→   642→
   643→   643→    return () => {
   644→   644→      cancelled = true;
   645→   645→      clearTimeout(safetyTimer);
   646→   646→    };
   647→   647→  }, [documentId, currentChapterIndex, currentChapterRefined, content, refineTriggered]);
   648→   648→
   649→   649→  /* ---------- scroll active chunk into view ---------- */
   650→   650→  const scrollRafRef = useRef<number | null>(null);

   660→   660→      }
   661→   661→    });
   662→   662→    return () => {
   663→   663→      if (scrollRafRef.current) cancelAnimationFrame(scrollRafRef.current);
   664→   664→    };
   665→   665→  }, [currentChapterIndex, activeChunkIndex, currentChapter]);
   666→   666→
   667→   667→  /* ---------- debounced progress patch ---------- */

   669→   669→  // otherwise re-fire this effect and patch again.

   685→   685→        lastChunkIndex: globalChunkIndex,

   695→   695→        // ignore network hiccups
   696→   696→      }

   699→   699→      if (patchTimer.current) clearTimeout(patchTimer.current);
   700→   700→    };

   703→   703→    readingProgress,
   704→   704→    documentId,
   705→   705→    totalChunks,
   706→   706→    setDoc,

   754→   754→    }

   768→   768→      // ignore
   769→   769→    }
   770→   770→  }, [documentId, globalChunkIndex, setDoc]);
   771→   771→
   772→   772→  /* ---------- keyboard navigation ---------- */

   776→   776→      const isTyping =
   777→   777→        !!target &&
   778→   778→        (target.tagName === "INPUT" ||
   779→   779→          target.tagName === "TEXTAREA" ||
