   709→   709→/* 5. Storage                                                          */
   710→   710→/* ------------------------------------------------------------------ */
   711→   711→
   712→   712→function StorageSection() {
   713→   713→  const { docs, loading, setDocs, refresh } = useDocuments();
   714→   714→  const [clearing, setClearing] = useState(false);
   715→   715→  const [deleting, setDeleting] = useState(false);
   716→   716→
   717→   717→  const stats = useMemo(() => {
   718→   718→    const total = docs.length;
   719→   719→    const bytes = docs.reduce((sum, d) => sum + (d.byteSize || 0), 0);
   720→   720→    const avg = total > 0 ? bytes / total : 0;
   721→   721→    return { total, bytes, avg };
   722→   722→  }, [docs]);
   723→   723→
   724→   724→  async function handleClearProgress() {
   725→   725→    if (docs.length === 0) {
   726→   726→      toast.info("Nothing to clear — your library is empty.");
   727→   727→      return;
   728→   728→    }
   729→   729→    setClearing(true);
   730→   730→    try {
   731→   731→      await Promise.all(
   732→   732→        docs.map((d) =>
   733→   733→          patchDocument(d.id, {
   734→   734→            readingProgress: 0,
   735→   735→            lastChunkIndex: 0,
   736→   736→            lastReadAt: null,
   737→   737→          }),
   738→   738→        ),
   739→   739→      );
   740→   740→      setDocs((prev) =>
   741→   741→        prev.map((d) => ({
   742→   742→          ...d,
   743→   743→          readingProgress: 0,
   744→   744→          lastChunkIndex: 0,
   745→   745→          lastReadAt: null,
   746→   746→        })),
   747→   747→      );
   748→   748→      toast.success(`Reading progress cleared for ${docs.length} document${docs.length === 1 ? "" : "s"}.`);
   749→   749→    } catch (e) {
   750→   750→      toast.error("Could not clear reading progress.", {
   751→   751→        description: e instanceof Error ? e.message : undefined,
   752→   752→      });
   753→   753→    } finally {
   754→   754→      setClearing(false);
   755→   755→    }
   756→   756→  }
   757→   757→
   758→   758→  async function handleDeleteAll() {
   759→   759→    if (docs.length === 0) {
   760→   760→      toast.info("Your library is already empty.");
   761→   761→      return;
   762→   762→    }
   763→   763→    setDeleting(true);
   764→   764→    try {
   765→   765→      await Promise.all(docs.map((d) => deleteDocument(d.id)));
   766→   766→      setDocs([]);
   767→   767→      toast.success(`Deleted ${docs.length} document${docs.length === 1 ? "" : "s"}.`);
   768→   768→      void refresh();
   769→   769→    } catch (e) {
   770→   770→      toast.error("Could not delete all documents.", {
   771→   771→        description: e instanceof Error ? e.message : undefined,
   772→   772→      });
   773→   773→      void refresh();
   774→   774→    } finally {
   775→   775→      setDeleting(false);
   776→   776→    }
   777→   777→  }
   778→   778→
   779→   779→  return (
   780→   780→    <>
   781→   781→      <div className="noir-card p-5">
   782→   782→        <p className="noir-eyebrow mb-3">Library</p>
   783→   783→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-3">
   784→   784→          <StatCell
   785→   785→            label="Total documents"
   786→   786→            value={loading ? "—" : String(stats.total)}
   787→   787→          />
   788→   788→          <StatCell
   789→   789→            label="Storage used"
   790→   790→            value={loading ? "—" : formatBytes(stats.bytes)}
   791→   791→          />
   792→   792→          <StatCell
   793→   793→            label="Average per doc"
   794→   794→            value={loading ? "—" : formatBytes(stats.avg)}
   795→   795→          />
   796→   796→        </div>
   797→   797→      </div>
   798→   798→
   799→   799→      <SubCard
   800→   800→        title="Maintenance"
   801→   801→        description="Reset reading state without removing documents, or wipe the library entirely."
   802→   802→      >
   803→   803→        <SettingRow
   804→   804→          label="Clear reading progress"
   805→   805→          hint="Resets last-read position, chunk index, and last-read time across all documents."
   806→   806→        >
   807→   807→          <Button
   808→   808→            type="button"
   809→   809→            variant="outline"
   810→   810→            className="noir-btn-ghost"
   811→   811→            disabled={clearing || deleting || loading || stats.total === 0}
   812→   812→            onClick={handleClearProgress}
   813→   813→          >
   814→   814→            {clearing ? (
   815→   815→              <Loader2 className="size-4 animate-spin" />
   816→   816→            ) : (
   817→   817→              <RotateCcw className="size-4" />
   818→   818→            )}
   819→   819→            Clear progress
   820→   820→          </Button>
   821→   821→        </SettingRow>
   822→   822→
   823→   823→        <SettingRow
   824→   824→          label="Delete all documents"
   825→   825→          hint="Permanently removes every document, its chapters, AI scenes, and activity events."
   826→   826→        >
   827→   827→          <AlertDialog>
   828→   828→            <AlertDialogTrigger asChild>
   829→   829→              <Button
   830→   830→                type="button"
   831→   831→                variant="destructive"
   832→   832→                className="border-red-500/40 bg-red-500/10 text-red-300 hover:bg-red-500/20"
   833→   833→                disabled={clearing || deleting || loading || stats.total === 0}
   834→   834→              >
   835→   835→                <Trash2 className="size-4" />
   836→   836→                {deleting ? "Deleting…" : "Delete all"}
   837→   837→              </Button>
   838→   838→            </AlertDialogTrigger>
   839→   839→            <AlertDialogContent>
   840→   840→              <AlertDialogHeader>
   841→   841→                <AlertDialogTitle className="flex items-center gap-2">
   842→   842→                  <AlertTriangle className="size-5 text-red-400" />
   843→   843→                  Delete all documents?
   844→   844→                </AlertDialogTitle>
   845→   845→                <AlertDialogDescription>
   846→   846→                  This permanently removes all {stats.total} document
   847→   847→                  {stats.total === 1 ? "" : "s"} from your library. This action
   848→   848→                  cannot be undone.
   849→   849→                </AlertDialogDescription>
   850→   850→              </AlertDialogHeader>
   851→   851→              <AlertDialogFooter>
   852→   852→                <AlertDialogCancel>Cancel</AlertDialogCancel>
   853→   853→                <AlertDialogAction
   854→   854→                  onClick={handleDeleteAll}
   855→   855→                  className="bg-red-600 text-white hover:bg-red-700"
   856→   856→                >
   857→   857→                  Delete everything
   858→   858→                </AlertDialogAction>
   859→   859→              </AlertDialogFooter>
   860→   860→            </AlertDialogContent>
   861→   861→          </AlertDialog>
   862→   862→        </SettingRow>
   863→   863→      </SubCard>
   864→   864→    </>
   865→   865→  );
