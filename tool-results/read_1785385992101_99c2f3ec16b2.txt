   375→/**
   376→ * Effective chunk count for a chapter. If the chapter has been OCR-refined,
   377→ * we split the refined text into ~1200-char chunks at paragraph boundaries.
   378→ * Otherwise we use the parser-produced chunks.
   379→ */
   380→function chapterChunkCount(ch: Chapter | undefined): number {
   381→  if (!ch) return 0;
   382→  if (ch.refinedText) {
   383→    // Split refined text into chunks for focus-mode navigation
   384→    const paras = ch.refinedText.split(/\n{2,}/).filter(Boolean);
   385→    return Math.max(1, Math.ceil(paras.length / 3));
   386→  }
   387→  return ch.chunks.length;
   388→}
   389→
   390→/** Split refined text into paragraph groups for rendering. */
   391→function refinedTextGroups(text: string): string[] {
   392→  const paras = text.split(/\n{2,}/).map((p) => p.trim()).filter(Boolean);
   393→  const groups: string[] = [];
   394→  for (let i = 0; i < paras.length; i += 3) {
   395→    groups.push(paras.slice(i, i + 3).join("\n\n"));
   396→  }
   397→  return groups;
   398→}

   611→    refineAbortRef.current = chIdx;
   612→
   613→    let cancelled = false;

   615→      .then((result) => {

   622→              i === chIdx ? { ...ch, refinedText: result.refinedText } : ch,
   623→            );
   624→            return { ...prev, chapters: newChapters };
   625→          });
   626→        }
   627→      })
   628→      .catch(() => {
   629→        // Best-effort — silently ignore (rate limits, network errors, etc.)
   630→      })
   631→      .finally(() => {
   632→        if (cancelled || refineAbortRef.current !== chIdx) return;
   633→        setRefining(false);
   634→      });
   635→
   636→    // Safety timeout — don't show the indicator for more than 60 seconds
   637→    const safetyTimer = setTimeout(() => {
   638→      if (!cancelled && refineAbortRef.current === chIdx) {
   639→        setRefining(false);
   640→      }
   641→    }, 60_000);
   642→
   643→    return () => {
   644→      cancelled = true;
   645→      clearTimeout(safetyTimer);
   646→    };

   695→        // ignore network hiccups

   699→      if (patchTimer.current) clearTimeout(patchTimer.current);
