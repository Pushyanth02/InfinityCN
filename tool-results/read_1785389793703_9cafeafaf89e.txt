   424→/**
   425→ * Effective chunk count for a chapter. If the chapter has been OCR-refined,
   426→ * we split the refined text into ~1200-char chunks at paragraph boundaries.
   427→ * Otherwise we use the parser-produced chunks.
   428→ */
   429→function chapterChunkCount(ch: Chapter | undefined): number {
   430→  if (!ch) return 0;
   431→  if (ch.refinedText) {
   432→    // Split refined text into chunks for focus-mode navigation
   433→    const paras = ch.refinedText.split(/\n{2,}/).filter(Boolean);
   434→    return Math.max(1, Math.ceil(paras.length / 3));
   435→  }
   436→  return ch.chunks.length;
   437→}
   438→
   439→/** Split refined text into paragraph groups for rendering. */
   440→function refinedTextGroups(text: string): string[] {
   441→  const paras = text.split(/\n{2,}/).map((p) => p.trim()).filter(Boolean);
   442→  const groups: string[] = [];
   443→  for (let i = 0; i < paras.length; i += 3) {
   444→    groups.push(paras.slice(i, i + 3).join("\n\n"));
   445→  }
   446→  return groups;
   447→}

   644→    };

   654→  /* ---------- per-chapter OCR refinement (background, non-blocking) ----------
   655→   * When a chapter is opened, it renders instantly with raw text. Then this
   656→   * effect triggers OCR refinement in the background. When refinement
   657→   * completes, the chapter's `refinedText` is updated and the reader
   658→   * re-renders with the cleaned text. The refinement is persisted server-side
   659→   * so the chapter is never reprocessed. */
   660→  const currentChapterRefined = currentChapter?.refinedText;
   661→  useEffect(() => {
   662→    if (!documentId || !content || !currentChapter) return;
   663→    const chIdx = currentChapterIndex;
   664→
   665→    // Already refined? Skip.
   666→    if (currentChapterRefined) return;
   667→    // Already triggered for this chapter? Skip.
   668→    if (refineTriggered.has(chIdx)) return;
   669→
   670→    // Mark as triggered
   671→    setRefineTriggered((prev) => new Set(prev).add(chIdx));
   672→    setRefining(true);
   673→    refineAbortRef.current = chIdx;
   674→
   675→    let cancelled = false;
   676→    refineChapter(documentId, chIdx, false)
   677→      .then((result) => {
   678→        if (cancelled || refineAbortRef.current !== chIdx) return;
   679→        if (result.refined && result.refinedText) {
   680→          // Update the chapter's refinedText in local state
   681→          setContent((prev) => {
   682→            if (!prev) return prev;
   683→            const newChapters = prev.chapters.map((ch, i) =>
   684→              i === chIdx ? { ...ch, refinedText: result.refinedText } : ch,
   685→            );
   686→            return { ...prev, chapters: newChapters };
   687→          });
   688→        }
   689→      })
   690→      .catch(() => {
   691→        // Best-effort — silently ignore (rate limits, network errors, etc.)
   692→      })
   693→      .finally(() => {
   694→        if (cancelled || refineAbortRef.current !== chIdx) return;
   695→        setRefining(false);
   696→      });

   699→    const safetyTimer = setTimeout(() => {
   700→      if (!cancelled && refineAbortRef.current === chIdx) {
   701→        setRefining(false);
   702→      }
   703→    }, 60_000);
   704→
   705→    return () => {
   706→      cancelled = true;
   707→      clearTimeout(safetyTimer);
   708→    };

   746→        readingProgress,
