   541→function AccessibilitySection({
   542→  settings,
   543→  update,
   544→}: {
   545→  settings: ReaderSettings;
   546→  update: (patch: Partial<ReaderSettings>) => void;
   547→}) {
   548→  const [focusRing, setFocusRing] = useState<"subtle" | "strong">("subtle");
   549→
   550→  // Sync the local focus-ring data attribute onto <html> so the global rule
   551→  // from globals.css can pick it up. We mirror the same pattern that
   552→  // useReaderSettings uses for contrast/motion/anim.
   553→  useEffect(() => {
   554→    if (typeof document === "undefined") return;
   555→    document.documentElement.dataset.focusRing = focusRing;
   556→  }, [focusRing]);
   557→
   558→  return (
   559→    <>
   560→      <SubCard
   561→        title="Contrast & motion"
   562→        description="Tune visibility of text and the strength of UI animation."
   563→      >
   564→        <SettingRow
   565→          label="Contrast"
   566→          hint="High contrast boosts text legibility against backgrounds."
   567→        >
   568→          <div className="flex gap-2">
   569→            {CONTRAST_OPTS.map((o) => (
   570→              <OptionButton
   571→                key={o.value}
   572→                active={settings.contrast === o.value}
   573→                onClick={() => update({ contrast: o.value })}
   574→              >
   575→                {o.label}
   576→              </OptionButton>
   577→            ))}
   578→          </div>
   579→        </SettingRow>
   580→
   581→        <SettingRow
   582→          label="Motion"
   583→          hint="Reduced motion disables decorative animation."
   584→        >
   585→          <div className="flex gap-2">
   586→            {MOTION_OPTS.map((o) => (
   587→              <OptionButton
   588→                key={o.value}
   589→                active={settings.motion === o.value}
   590→                onClick={() => update({ motion: o.value })}
   591→              >
   592→                {o.label}
   593→              </OptionButton>
   594→            ))}
   595→          </div>
   596→        </SettingRow>
   597→
   598→        <SettingRow
   599→          label="Animation speed"
   600→          hint="Off disables transitions entirely; Fast / Slow scale durations."
   601→        >
   602→          <div className="flex flex-wrap gap-2">
   603→            {ANIM_OPTS.map((o) => (
   604→              <OptionButton
   605→                key={o.value}
   606→                active={settings.animSpeed === o.value}
   607→                onClick={() => update({ animSpeed: o.value })}
   608→              >
   609→                {o.label}
   610→              </OptionButton>
   611→            ))}
   612→          </div>
   613→        </SettingRow>
   614→
   615→        <SettingRow
   616→          label="Focus ring strength"
   617→          hint="Make focus outlines more prominent for keyboard navigation."
   618→        >
   619→          <div className="flex gap-2">
   620→            {(["subtle", "strong"] as const).map((v) => (
   621→              <OptionButton
   622→                key={v}
   623→                active={focusRing === v}
   624→                onClick={() => setFocusRing(v)}
   625→              >
   626→                {v === "subtle" ? "Subtle" : "Strong"}
   627→              </OptionButton>
   628→            ))}
   629→          </div>
   630→        </SettingRow>
   631→      </SubCard>
   632→
   633→      <SubCard
   634→        title="Keyboard"
   635→        description="Show in-context hints for the keyboard shortcuts available across the app."
   636→      >
   637→        <SettingRow
   638→          label="Keyboard hints"
   639→          hint="Reveal ⌘K, arrows, F, Esc labels where relevant."
   640→        >
   641→          <Switch
   642→            checked={settings.kbdHints}
   643→            onCheckedChange={(v) => update({ kbdHints: v })}
   644→            aria-label="Toggle keyboard hints"
   645→          />
   646→        </SettingRow>
   647→      </SubCard>
   648→    </>
   649→  );

   708→/* ------------------------------------------------------------------ */
   709→/* 5. Storage                                                          */
   710→/* ------------------------------------------------------------------ */
   711→
   712→function StorageSection() {
   713→  const { docs, loading, setDocs, refresh } = useDocuments();
   714→  const [clearing, setClearing] = useState(false);
   715→  const [deleting, setDeleting] = useState(false);
   716→
   717→  const stats = useMemo(() => {
   718→    const total = docs.length;
   719→    const bytes = docs.reduce((sum, d) => sum + (d.byteSize || 0), 0);
   720→    const avg = total > 0 ? bytes / total : 0;
   721→    return { total, bytes, avg };
   722→  }, [docs]);
   723→
   724→  async function handleClearProgress() {
   725→    if (docs.length === 0) {
   726→      toast.info("Nothing to clear — your library is empty.");
   727→      return;
   728→    }
   729→    setClearing(true);
   730→    try {
   731→      await Promise.all(
   732→        docs.map((d) =>
   733→          patchDocument(d.id, {
   734→            readingProgress: 0,
   735→            lastChunkIndex: 0,
   736→            lastReadAt: null,
   737→          }),
   738→        ),
   739→      );
   740→      setDocs((prev) =>
   741→        prev.map((d) => ({
   742→          ...d,
   743→          readingProgress: 0,
   744→          lastChunkIndex: 0,
   745→          lastReadAt: null,
   746→        })),
   747→      );
   748→      toast.success(`Reading progress cleared for ${docs.length} document${docs.length === 1 ? "" : "s"}.`);
   749→    } catch (e) {
   750→      toast.error("Could not clear reading progress.", {
   751→        description: e instanceof Error ? e.message : undefined,
   752→      });
   753→    } finally {
   754→      setClearing(false);
   755→    }
   756→  }
   757→
   758→  async function handleDeleteAll() {
   759→    if (docs.length === 0) {
   760→      toast.info("Your library is already empty.");
   761→      return;
   762→    }
   763→    setDeleting(true);
   764→    try {
   765→      await Promise.all(docs.map((d) => deleteDocument(d.id)));
   766→      setDocs([]);
   767→      toast.success(`Deleted ${docs.length} document${docs.length === 1 ? "" : "s"}.`);
   768→      void refresh();
   769→    } catch (e) {
   770→      toast.error("Could not delete all documents.", {
   771→        description: e instanceof Error ? e.message : undefined,
   772→      });
   773→      void refresh();
   774→    } finally {
   775→      setDeleting(false);
   776→    }
   777→  }
   778→
   779→  return (
   780→    <>
   781→      <div className="noir-card p-5">
   782→        <p className="noir-eyebrow mb-3">Library</p>
   783→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-3">
   784→          <StatCell
   785→            label="Total documents"
   786→            value={loading ? "—" : String(stats.total)}
   787→          />
   788→          <StatCell
   789→            label="Storage used"
   790→            value={loading ? "—" : formatBytes(stats.bytes)}
   791→          />
   792→          <StatCell
   793→            label="Average per doc"
   794→            value={loading ? "—" : formatBytes(stats.avg)}
   795→          />
   796→        </div>
   797→      </div>
   798→
   799→      <SubCard
   800→        title="Maintenance"
   801→        description="Reset reading state without removing documents, or wipe the library entirely."
   802→      >
   803→        <SettingRow
   804→          label="Clear reading progress"
   805→          hint="Resets last-read position, chunk index, and last-read time across all documents."
   806→        >
   807→          <Button
   808→            type="button"
   809→            variant="outline"
   810→            className="noir-btn-ghost"
   811→            disabled={clearing || deleting || loading || stats.total === 0}
   812→            onClick={handleClearProgress}
   813→          >
   814→            {clearing ? (
   815→              <Loader2 className="size-4 animate-spin" />
   816→            ) : (
   817→              <RotateCcw className="size-4" />
   818→            )}
   819→            Clear progress
   820→          </Button>
   821→        </SettingRow>
   822→
   823→        <SettingRow
   824→          label="Delete all documents"
   825→          hint="Permanently removes every document, its chapters, AI scenes, and activity events."
   826→        >
   827→          <AlertDialog>
   828→            <AlertDialogTrigger asChild>
   829→              <Button
   830→                type="button"
   831→                variant="destructive"
   832→                className="border-red-500/40 bg-red-500/10 text-red-300 hover:bg-red-500/20"
   833→                disabled={clearing || deleting || loading || stats.total === 0}
   834→              >
   835→                <Trash2 className="size-4" />
   836→                {deleting ? "Deleting…" : "Delete all"}
   837→              </Button>
   838→            </AlertDialogTrigger>
   839→            <AlertDialogContent>
   840→              <AlertDialogHeader>
   841→                <AlertDialogTitle className="flex items-center gap-2">
   842→                  <AlertTriangle className="size-5 text-red-400" />
   843→                  Delete all documents?
   844→                </AlertDialogTitle>
   845→                <AlertDialogDescription>
   846→                  This permanently removes all {stats.total} document
   847→                  {stats.total === 1 ? "" : "s"} from your library. This action
   848→                  cannot be undone.
   849→                </AlertDialogDescription>
   850→              </AlertDialogHeader>
   851→              <AlertDialogFooter>
   852→                <AlertDialogCancel>Cancel</AlertDialogCancel>
   853→                <AlertDialogAction
   854→                  onClick={handleDeleteAll}
   855→                  className="bg-red-600 text-white hover:bg-red-700"
   856→                >
   857→                  Delete everything
   858→                </AlertDialogAction>
   859→              </AlertDialogFooter>
   860→            </AlertDialogContent>
   861→          </AlertDialog>
   862→        </SettingRow>
   863→      </SubCard>
   864→    </>
   865→  );
   866→}
